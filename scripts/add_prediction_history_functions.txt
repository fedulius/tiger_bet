CREATE OR REPLACE FUNCTION public.analysis_type_resolve(
    in_analysis_type_code varchar
)
RETURNS integer
LANGUAGE plpgsql
AS $function$
DECLARE
    out_analysis_type_id integer;
BEGIN
    SELECT analysis_type_id
    FROM public.analysis_type
    WHERE analysis_type_code = in_analysis_type_code
      AND is_active = true
    INTO out_analysis_type_id;

    IF out_analysis_type_id IS NULL THEN
        RAISE EXCEPTION 'Unknown or inactive analysis_type_code: %', in_analysis_type_code
            USING ERRCODE = 'foreign_key_violation';
    END IF;

    RETURN out_analysis_type_id;
END;
$function$;

CREATE OR REPLACE FUNCTION public.match_analysis_type_set(
    in_match_analysis_id integer,
    in_analysis_type_code varchar
)
RETURNS integer
LANGUAGE plpgsql
AS $function$
DECLARE
    tmp_analysis_type_id integer;
BEGIN
    tmp_analysis_type_id := public.analysis_type_resolve(in_analysis_type_code);

    UPDATE public.match_analysis
    SET analysis_type_id = tmp_analysis_type_id,
        analysis_update_at = now()
    WHERE match_analysis_id = in_match_analysis_id;

    IF NOT FOUND THEN
        RAISE EXCEPTION 'match_analysis not found: %', in_match_analysis_id
            USING ERRCODE = 'no_data_found';
    END IF;

    RETURN in_match_analysis_id;
END;
$function$;

CREATE OR REPLACE FUNCTION bet.prediction_card_create(
    in_card_type_code varchar,
    in_card_status_code varchar DEFAULT 'published',
    in_primary_match_id integer DEFAULT NULL,
    in_primary_match_analysis_id integer DEFAULT NULL,
    in_published_at timestamptz DEFAULT now(),
    in_published_date date DEFAULT NULL,
    in_edition_no integer DEFAULT 1,
    in_title text DEFAULT NULL,
    in_headline text DEFAULT NULL,
    in_brief text DEFAULT NULL,
    in_risk_note text DEFAULT NULL,
    in_visibility_scope varchar DEFAULT 'public',
    in_publication_hash varchar DEFAULT NULL,
    in_snapshot jsonb DEFAULT '{}'::jsonb
)
RETURNS bigint
LANGUAGE plpgsql
AS $function$
DECLARE
    tmp_card_type_id smallint;
    tmp_card_status_id smallint;
    tmp_publication_hash varchar;
    out_prediction_card_id bigint;
BEGIN
    tmp_card_type_id := bet.card_type_resolve(in_card_type_code);
    tmp_card_status_id := bet.card_status_resolve(in_card_status_code);

    tmp_publication_hash := COALESCE(
        in_publication_hash,
        md5(concat_ws('|',
            in_card_type_code,
            COALESCE(in_primary_match_id::text, ''),
            COALESCE(in_primary_match_analysis_id::text, ''),
            COALESCE(in_edition_no::text, '1'),
            COALESCE(in_published_at::text, now()::text),
            COALESCE(in_snapshot::text, '{}')
        ))
    );

    INSERT INTO bet.prediction_card(
          card_type_id
        , card_status_id
        , primary_match_id
        , primary_match_analysis_id
        , published_at
        , published_date
        , edition_no
        , title
        , headline
        , brief
        , risk_note
        , visibility_scope
        , publication_hash
        , snapshot
    )
    VALUES (
          tmp_card_type_id
        , tmp_card_status_id
        , in_primary_match_id
        , in_primary_match_analysis_id
        , in_published_at
        , COALESCE(in_published_date, (in_published_at AT TIME ZONE 'Europe/Moscow')::date)
        , COALESCE(in_edition_no, 1)
        , in_title
        , in_headline
        , in_brief
        , in_risk_note
        , COALESCE(in_visibility_scope, 'public')
        , tmp_publication_hash
        , COALESCE(in_snapshot, '{}'::jsonb)
    )
    RETURNING prediction_card_id
    INTO out_prediction_card_id;

    RETURN out_prediction_card_id;
END;
$function$;

-- Convenience function: create a daily/single card from existing match_analysis.
CREATE OR REPLACE FUNCTION bet.prediction_card_create_from_match_analysis(
    in_match_analysis_id integer,
    in_card_type_code varchar DEFAULT 'daily',
    in_card_status_code varchar DEFAULT 'published',
    in_visibility_scope varchar DEFAULT 'public',
    in_published_at timestamptz DEFAULT now(),
    in_edition_no integer DEFAULT 1
)
RETURNS bigint
LANGUAGE plpgsql
AS $function$
DECLARE
    tmp_match_id integer;
    tmp_analysis public.match_analysis%ROWTYPE;
    tmp_snapshot jsonb;
    tmp_publication_hash varchar;
    out_prediction_card_id bigint;
BEGIN
    SELECT ma.*
    FROM public.match_analysis ma
    WHERE ma.match_analysis_id = in_match_analysis_id
    INTO tmp_analysis;

    IF tmp_analysis.match_analysis_id IS NULL THEN
        RAISE EXCEPTION 'match_analysis not found: %', in_match_analysis_id
            USING ERRCODE = 'no_data_found';
    END IF;

    SELECT match_id
    FROM public.match_source
    WHERE match_source_id = tmp_analysis.match_source_id
    INTO tmp_match_id;

    IF tmp_match_id IS NULL THEN
        RAISE EXCEPTION 'match_source/match not found for match_analysis: %', in_match_analysis_id
            USING ERRCODE = 'foreign_key_violation';
    END IF;

    tmp_snapshot := jsonb_build_object(
        'match_analysis_id', tmp_analysis.match_analysis_id,
        'match_source_id', tmp_analysis.match_source_id,
        'analysis_hash', tmp_analysis.analysis_hash,
        'analysis_headline', tmp_analysis.analysis_headline,
        'analysis_brief', tmp_analysis.analysis_brief,
        'analysis_risk_note', tmp_analysis.analysis_risk_note,
        'recommended_bets', tmp_analysis.recommended_bets,
        'model_name', tmp_analysis.model_name,
        'prompt_version', tmp_analysis.prompt_version,
        'analysis_create_at', tmp_analysis.analysis_create_at
    );

    tmp_publication_hash := md5(concat_ws('|',
        in_card_type_code,
        tmp_analysis.match_analysis_id::text,
        COALESCE(tmp_analysis.analysis_hash, ''),
        COALESCE(in_edition_no::text, '1')
    ));

    out_prediction_card_id := bet.prediction_card_create(
          in_card_type_code => in_card_type_code
        , in_card_status_code => in_card_status_code
        , in_primary_match_id => tmp_match_id
        , in_primary_match_analysis_id => tmp_analysis.match_analysis_id
        , in_published_at => in_published_at
        , in_published_date => (in_published_at AT TIME ZONE 'Europe/Moscow')::date
        , in_edition_no => in_edition_no
        , in_title => tmp_analysis.analysis_headline
        , in_headline => tmp_analysis.analysis_headline
        , in_brief => tmp_analysis.analysis_brief
        , in_risk_note => tmp_analysis.analysis_risk_note
        , in_visibility_scope => in_visibility_scope
        , in_publication_hash => tmp_publication_hash
        , in_snapshot => tmp_snapshot
    );

    PERFORM bet.prediction_card_source_create(
          in_prediction_card_id => out_prediction_card_id
        , in_match_analysis_id => tmp_analysis.match_analysis_id
        , in_source_role => 'primary'
        , in_ordinal => 1
    );

    RETURN out_prediction_card_id;
END;
$function$;

CREATE OR REPLACE FUNCTION bet.prediction_bet_create(
    in_prediction_card_id bigint,
    in_kind varchar DEFAULT 'single',
    in_ordinal smallint DEFAULT 1,
    in_match_id integer DEFAULT NULL,
    in_match_analysis_id integer DEFAULT NULL,
    in_market_type_code varchar DEFAULT NULL,
    in_period_code varchar DEFAULT 'full_time',
    in_participant_scope varchar DEFAULT NULL,
    in_selection_code varchar DEFAULT NULL,
    in_line_value numeric DEFAULT NULL,
    in_score_home smallint DEFAULT NULL,
    in_score_away smallint DEFAULT NULL,
    in_market_key text DEFAULT NULL,
    in_display_label text DEFAULT NULL,
    in_odds_decimal numeric DEFAULT NULL,
    in_risk_level_code varchar DEFAULT NULL,
    in_confidence numeric DEFAULT NULL,
    in_reason text DEFAULT NULL,
    in_source_payload jsonb DEFAULT NULL,
    in_selection_snapshot jsonb DEFAULT '{}'::jsonb
)
RETURNS bigint
LANGUAGE plpgsql
AS $function$
DECLARE
    tmp_market_type_id smallint;
    tmp_period_id smallint;
    tmp_risk_level_id smallint;
    tmp_market_key text;
    out_prediction_bet_id bigint;
BEGIN
    IF COALESCE(in_kind, 'single') = 'single' THEN
        tmp_market_type_id := bet.market_type_resolve(in_market_type_code);
        tmp_period_id := bet.period_resolve(COALESCE(in_period_code, 'full_time'));
        tmp_market_key := COALESCE(
            in_market_key,
            concat_ws('|', in_market_type_code, COALESCE(in_period_code, 'full_time'), in_participant_scope, in_selection_code, in_line_value::text)
        );
    ELSE
        tmp_market_type_id := NULL;
        tmp_period_id := NULL;
        tmp_market_key := COALESCE(in_market_key, concat_ws('|', 'express', in_prediction_card_id::text, in_ordinal::text));
    END IF;

    tmp_risk_level_id := bet.risk_level_resolve(in_risk_level_code);

    INSERT INTO bet.prediction_bet(
          prediction_card_id
        , kind
        , ordinal
        , match_id
        , match_analysis_id
        , market_type_id
        , period_id
        , participant_scope
        , selection_code
        , line_value
        , score_home
        , score_away
        , market_key
        , display_label
        , odds_decimal
        , risk_level_id
        , confidence
        , reason
        , source_payload
        , selection_snapshot
    )
    VALUES (
          in_prediction_card_id
        , COALESCE(in_kind, 'single')
        , COALESCE(in_ordinal, 1)
        , in_match_id
        , in_match_analysis_id
        , tmp_market_type_id
        , tmp_period_id
        , in_participant_scope
        , in_selection_code
        , in_line_value
        , in_score_home
        , in_score_away
        , tmp_market_key
        , COALESCE(in_display_label, tmp_market_key)
        , in_odds_decimal
        , tmp_risk_level_id
        , in_confidence
        , in_reason
        , in_source_payload
        , COALESCE(in_selection_snapshot, '{}'::jsonb)
    )
    RETURNING prediction_bet_id
    INTO out_prediction_bet_id;

    RETURN out_prediction_bet_id;
END;
$function$;

CREATE OR REPLACE FUNCTION bet.prediction_bet_leg_create(
    in_prediction_bet_id bigint,
    in_ordinal smallint,
    in_match_id integer,
    in_match_analysis_id integer DEFAULT NULL,
    in_market_type_code varchar DEFAULT NULL,
    in_period_code varchar DEFAULT 'full_time',
    in_participant_scope varchar DEFAULT 'match',
    in_selection_code varchar DEFAULT NULL,
    in_line_value numeric DEFAULT NULL,
    in_score_home smallint DEFAULT NULL,
    in_score_away smallint DEFAULT NULL,
    in_market_key text DEFAULT NULL,
    in_display_label text DEFAULT NULL,
    in_odds_decimal numeric DEFAULT NULL,
    in_leg_snapshot jsonb DEFAULT '{}'::jsonb
)
RETURNS bigint
LANGUAGE plpgsql
AS $function$
DECLARE
    tmp_parent_kind varchar;
    tmp_market_type_id smallint;
    tmp_period_id smallint;
    tmp_market_key text;
    out_prediction_bet_leg_id bigint;
BEGIN
    SELECT kind
    FROM bet.prediction_bet
    WHERE prediction_bet_id = in_prediction_bet_id
    INTO tmp_parent_kind;

    IF tmp_parent_kind IS NULL THEN
        RAISE EXCEPTION 'prediction_bet not found: %', in_prediction_bet_id
            USING ERRCODE = 'no_data_found';
    END IF;

    IF tmp_parent_kind <> 'express' THEN
        RAISE EXCEPTION 'prediction_bet_leg can be attached only to kind=express, got: %', tmp_parent_kind
            USING ERRCODE = 'check_violation';
    END IF;

    tmp_market_type_id := bet.market_type_resolve(in_market_type_code);
    tmp_period_id := bet.period_resolve(COALESCE(in_period_code, 'full_time'));
    tmp_market_key := COALESCE(
        in_market_key,
        concat_ws('|', in_market_type_code, COALESCE(in_period_code, 'full_time'), in_participant_scope, in_selection_code, in_line_value::text)
    );

    INSERT INTO bet.prediction_bet_leg(
          prediction_bet_id
        , ordinal
        , match_id
        , match_analysis_id
        , market_type_id
        , period_id
        , participant_scope
        , selection_code
        , line_value
        , score_home
        , score_away
        , market_key
        , display_label
        , odds_decimal
        , leg_snapshot
    )
    VALUES (
          in_prediction_bet_id
        , in_ordinal
        , in_match_id
        , in_match_analysis_id
        , tmp_market_type_id
        , tmp_period_id
        , COALESCE(in_participant_scope, 'match')
        , in_selection_code
        , in_line_value
        , in_score_home
        , in_score_away
        , tmp_market_key
        , COALESCE(in_display_label, tmp_market_key)
        , in_odds_decimal
        , COALESCE(in_leg_snapshot, '{}'::jsonb)
    )
    RETURNING prediction_bet_leg_id
    INTO out_prediction_bet_leg_id;

    RETURN out_prediction_bet_leg_id;
END;
$function$;

CREATE OR REPLACE FUNCTION bet.bet_settlement_upsert(
    in_prediction_bet_id bigint,
    in_settlement_status_code varchar,
    in_settlement_result_code varchar DEFAULT NULL,
    in_settled_at timestamptz DEFAULT NULL,
    in_rule_version varchar DEFAULT 'v1',
    in_score_scope jsonb DEFAULT '{}'::jsonb,
    in_outcome_value jsonb DEFAULT NULL,
    in_profit_factor numeric DEFAULT NULL,
    in_reason_code varchar DEFAULT NULL,
    in_reason_text text DEFAULT NULL,
    in_calculated_by varchar DEFAULT 'rule_engine',
    in_raw_payload jsonb DEFAULT NULL,
    in_write_revision boolean DEFAULT true
)
RETURNS bigint
LANGUAGE plpgsql
AS $function$
DECLARE
    tmp_status_id smallint;
    tmp_result_id smallint;
    old_row bet.bet_settlement%ROWTYPE;
    out_bet_settlement_id bigint;
BEGIN
    tmp_status_id := bet.settlement_status_resolve(in_settlement_status_code);
    tmp_result_id := bet.settlement_result_resolve(in_settlement_result_code);

    SELECT *
    FROM bet.bet_settlement
    WHERE prediction_bet_id = in_prediction_bet_id
    INTO old_row;

    IF old_row.bet_settlement_id IS NULL THEN
        INSERT INTO bet.bet_settlement(
              prediction_bet_id
            , settlement_status_id
            , settlement_result_id
            , settled_at
            , settlement_version
            , rule_version
            , score_scope
            , outcome_value
            , profit_factor
            , reason_code
            , reason_text
            , calculated_at
            , calculated_by
            , raw_payload
        )
        VALUES (
              in_prediction_bet_id
            , tmp_status_id
            , tmp_result_id
            , in_settled_at
            , 1
            , COALESCE(in_rule_version, 'v1')
            , COALESCE(in_score_scope, '{}'::jsonb)
            , in_outcome_value
            , in_profit_factor
            , in_reason_code
            , in_reason_text
            , now()
            , COALESCE(in_calculated_by, 'rule_engine')
            , in_raw_payload
        )
        RETURNING bet_settlement_id
        INTO out_bet_settlement_id;

        IF in_write_revision THEN
            INSERT INTO bet.bet_settlement_revision(
                  prediction_bet_id
                , previous_settlement_status_id
                , previous_settlement_result_id
                , new_settlement_status_id
                , new_settlement_result_id
                , previous_profit_factor
                , new_profit_factor
                , reason_code
                , reason_text
                , changed_by
                , payload
            )
            VALUES (
                  in_prediction_bet_id
                , NULL
                , NULL
                , tmp_status_id
                , tmp_result_id
                , NULL
                , in_profit_factor
                , COALESCE(in_reason_code, 'initial_settlement')
                , in_reason_text
                , COALESCE(in_calculated_by, 'rule_engine')
                , COALESCE(in_raw_payload, '{}'::jsonb)
            );
        END IF;
    ELSE
        UPDATE bet.bet_settlement
        SET settlement_status_id = tmp_status_id,
            settlement_result_id = tmp_result_id,
            settled_at = in_settled_at,
            settlement_version = old_row.settlement_version + 1,
            rule_version = COALESCE(in_rule_version, old_row.rule_version),

            profit_factor = in_profit_factor,
            reason_code = in_reason_code,
            reason_text = in_reason_text,
            calculated_at = now(),
            calculated_by = COALESCE(in_calculated_by, 'rule_engine'),
            raw_payload = in_raw_payload
        WHERE bet_settlement_id = old_row.bet_settlement_id
        RETURNING bet_settlement_id
        INTO out_bet_settlement_id;

        IF in_write_revision THEN
            INSERT INTO bet.bet_settlement_revision(
                  prediction_bet_id
                , previous_settlement_status_id
                , previous_settlement_result_id
                , new_settlement_status_id
                , new_settlement_result_id
                , previous_profit_factor
                , new_profit_factor
                , reason_code
                , reason_text
                , changed_by
                , payload
            )
            VALUES (
                  in_prediction_bet_id
                , old_row.settlement_status_id
                , old_row.settlement_result_id
                , tmp_status_id
                , tmp_result_id
                , old_row.profit_factor
                , in_profit_factor
                , in_reason_code
                , in_reason_text
                , COALESCE(in_calculated_by, 'rule_engine')
                , COALESCE(in_raw_payload, '{}'::jsonb)
            );
        END IF;
    END IF;

    RETURN out_bet_settlement_id;
END;
$function$;

CREATE OR REPLACE FUNCTION bet.bet_leg_settlement_upsert(
    in_prediction_bet_leg_id bigint,
    in_settlement_status_code varchar,
    in_settlement_result_code varchar DEFAULT NULL,
    in_settled_at timestamptz DEFAULT NULL,
    in_rule_version varchar DEFAULT 'v1',
    in_score_scope jsonb DEFAULT '{}'::jsonb,
    in_outcome_value jsonb DEFAULT NULL,
    in_profit_factor numeric DEFAULT NULL,
    in_reason_code varchar DEFAULT NULL,
    in_reason_text text DEFAULT NULL,
    in_calculated_by varchar DEFAULT 'rule_engine',
    in_raw_payload jsonb DEFAULT NULL,
    in_write_revision boolean DEFAULT true
)
RETURNS bigint
LANGUAGE plpgsql
AS $function$
DECLARE
    tmp_status_id smallint;
    tmp_result_id smallint;
    old_row bet.bet_leg_settlement%ROWTYPE;
    out_bet_leg_settlement_id bigint;
BEGIN
    tmp_status_id := bet.settlement_status_resolve(in_settlement_status_code);
    tmp_result_id := bet.settlement_result_resolve(in_settlement_result_code);

    SELECT *
    FROM bet.bet_leg_settlement
    WHERE prediction_bet_leg_id = in_prediction_bet_leg_id
    INTO old_row;

    IF old_row.bet_leg_settlement_id IS NULL THEN
        INSERT INTO bet.bet_leg_settlement(
              prediction_bet_leg_id
            , settlement_status_id
            , settlement_result_id
            , settled_at
            , settlement_version
            , rule_version
            , score_scope
            , outcome_value
            , profit_factor
            , reason_code
            , reason_text
            , calculated_at
            , calculated_by
            , raw_payload
        )
        VALUES (
              in_prediction_bet_leg_id
            , tmp_status_id
            , tmp_result_id
            , in_settled_at
            , 1
            , COALESCE(in_rule_version, 'v1')
            , COALESCE(in_score_scope, '{}'::jsonb)
            , in_outcome_value
            , in_profit_factor
            , in_reason_code
            , in_reason_text
            , now()
            , COALESCE(in_calculated_by, 'rule_engine')
            , in_raw_payload
        )
        RETURNING bet_leg_settlement_id
        INTO out_bet_leg_settlement_id;

        IF in_write_revision THEN
            INSERT INTO bet.bet_settlement_revision(
                  prediction_bet_leg_id
                , previous_settlement_status_id
                , previous_settlement_result_id
                , new_settlement_status_id
                , new_settlement_result_id
                , previous_profit_factor
                , new_profit_factor
                , reason_code
                , reason_text
                , changed_by
                , payload
            )
            VALUES (
                  in_prediction_bet_leg_id
                , NULL
                , NULL
                , tmp_status_id
                , tmp_result_id
                , NULL
                , in_profit_factor
                , COALESCE(in_reason_code, 'initial_leg_settlement')
                , in_reason_text
                , COALESCE(in_calculated_by, 'rule_engine')
                , COALESCE(in_raw_payload, '{}'::jsonb)
            );
        END IF;
    ELSE
        UPDATE bet.bet_leg_settlement
        SET settlement_status_id = tmp_status_id,
            settlement_result_id = tmp_result_id,
            settled_at = in_settled_at,
            settlement_version = old_row.settlement_version + 1,
            rule_version = COALESCE(in_rule_version, old_row.rule_version),
            score_scope = COALESCE(in_score_scope, '{}'::jsonb),
            outcome_value = in_outcome_value,
            profit_factor = in_profit_factor,
            reason_code = in_reason_code,
            reason_text = in_reason_text,
            calculated_at = now(),
            calculated_by = COALESCE(in_calculated_by, 'rule_engine'),
            raw_payload = in_raw_payload
        WHERE bet_leg_settlement_id = old_row.bet_leg_settlement_id
        RETURNING bet_leg_settlement_id
        INTO out_bet_leg_settlement_id;

        IF in_write_revision THEN
            INSERT INTO bet.bet_settlement_revision(
                  prediction_bet_leg_id
                , previous_settlement_status_id
                , previous_settlement_result_id
                , new_settlement_status_id
                , new_settlement_result_id
                , previous_profit_factor
                , new_profit_factor
                , reason_code
                , reason_text
                , changed_by
                , payload
            )
            VALUES (
                  in_prediction_bet_leg_id
                , old_row.settlement_status_id
                , old_row.settlement_result_id
                , tmp_status_id
                , tmp_result_id
                , old_row.profit_factor
                , in_profit_factor
                , in_reason_code
                , in_reason_text
                , COALESCE(in_calculated_by, 'rule_engine')
                , COALESCE(in_raw_payload, '{}'::jsonb)
            );
        END IF;
    END IF;

    RETURN out_bet_leg_settlement_id;
END;
$function$;

-- Atomic/idempotent one-analysis backfill. One function call is one transaction.
DROP FUNCTION IF EXISTS bet.prediction_history_backfill_one(jsonb);
CREATE OR REPLACE FUNCTION bet.prediction_history_backfill_one(in_payload jsonb)
RETURNS TABLE(out_prediction_card_id bigint, bets_count integer, skipped boolean)
LANGUAGE plpgsql
AS $function$
DECLARE old_card bigint; old_bets integer; old_settlements integer; expected integer := jsonb_array_length(COALESCE(in_payload->'normalized_bets','[]'::jsonb)); item jsonb; market_id smallint; period_id_value smallint; risk_id smallint; bet_id bigint; new_card bigint;
BEGIN
  SELECT pc.prediction_card_id INTO old_card FROM bet.prediction_card pc WHERE pc.primary_match_analysis_id=(in_payload->>'match_analysis_id')::integer OR pc.publication_hash=in_payload->>'publication_hash' ORDER BY pc.prediction_card_id LIMIT 1;
  IF old_card IS NOT NULL THEN
    SELECT COUNT(*) INTO old_bets FROM bet.prediction_bet pb WHERE pb.prediction_card_id=old_card;
    SELECT COUNT(*) INTO old_settlements FROM bet.bet_settlement bs JOIN bet.prediction_bet pb ON pb.prediction_bet_id=bs.prediction_bet_id WHERE pb.prediction_card_id=old_card;
    IF old_bets=expected AND old_settlements=expected AND EXISTS (SELECT 1 FROM bet.prediction_card_source pcs WHERE pcs.prediction_card_id=old_card AND pcs.match_analysis_id=(in_payload->>'match_analysis_id')::integer) THEN RETURN QUERY SELECT old_card,old_bets,true; RETURN; END IF;
    DELETE FROM bet.bet_settlement bs USING bet.prediction_bet pb WHERE bs.prediction_bet_id=pb.prediction_bet_id AND pb.prediction_card_id=old_card;
    DELETE FROM bet.prediction_bet pb WHERE pb.prediction_card_id=old_card; DELETE FROM bet.prediction_card_source pcs WHERE pcs.prediction_card_id=old_card; DELETE FROM bet.prediction_card pc WHERE pc.prediction_card_id=old_card;
  END IF;
  INSERT INTO bet.prediction_card(card_type_id,card_status_id,primary_match_id,primary_match_analysis_id,published_at,published_date,edition_no,title,headline,brief,risk_note,visibility_scope,publication_hash,snapshot)
  SELECT ct.card_type_id,cs.card_status_id,(in_payload->>'match_id')::integer,(in_payload->>'match_analysis_id')::integer,(in_payload->>'published_at')::timestamptz,((in_payload->>'published_at')::timestamptz AT TIME ZONE 'Europe/Moscow')::date,1,in_payload->>'analysis_headline',in_payload->>'analysis_headline',in_payload->>'analysis_brief',in_payload->>'analysis_risk_note','public',in_payload->>'publication_hash',in_payload
  FROM bet.card_type ct CROSS JOIN bet.card_status cs WHERE ct.card_type_code='daily' AND ct.is_active AND cs.card_status_code='published' AND cs.is_active RETURNING prediction_card_id INTO new_card;
  INSERT INTO bet.prediction_card_source(prediction_card_id,match_analysis_id,source_role,ordinal) VALUES(new_card,(in_payload->>'match_analysis_id')::integer,'primary',1);
  FOR item IN SELECT value FROM jsonb_array_elements(in_payload->'normalized_bets') LOOP
    SELECT market_type_id INTO market_id FROM bet.market_type WHERE market_type_code=item->>'market_type_code' AND is_active; SELECT period_id INTO period_id_value FROM bet.period WHERE period_code=item->>'period_code' AND is_active; SELECT risk_level_id INTO risk_id FROM bet.risk_level WHERE risk_level_code=item->>'risk_level_code' AND is_active;
    INSERT INTO bet.prediction_bet(prediction_card_id,kind,ordinal,match_id,match_analysis_id,market_type_id,period_id,participant_scope,selection_code,line_value,score_home,score_away,market_key,display_label,odds_decimal,risk_level_id,confidence,reason,source_payload,selection_snapshot)
    VALUES(new_card,'single',(item->>'ordinal')::smallint,(in_payload->>'match_id')::integer,(in_payload->>'match_analysis_id')::integer,market_id,period_id_value,item->>'participant_scope',item->>'selection_code',NULLIF(item->>'line_value','')::numeric,NULLIF(item->>'score_home','')::smallint,NULLIF(item->>'score_away','')::smallint,item->>'market_key',item->>'display_label',NULLIF(item->>'odds_decimal','')::numeric,risk_id,NULLIF(item->>'confidence','')::numeric,item->>'reason',item->'source_payload',item->'selection_snapshot') RETURNING prediction_bet_id INTO bet_id;
    INSERT INTO bet.bet_settlement(prediction_bet_id,settlement_status_id,calculated_by,reason_code,reason_text) SELECT bet_id,settlement_status_id,'backfill','awaiting_settlement','Ожидает расчёта результата' FROM bet.settlement_status WHERE settlement_status_code='pending' AND is_active;
  END LOOP; RETURN QUERY SELECT new_card,expected,false;
END; $function$;

CREATE OR REPLACE FUNCTION bet.repair_correct_score_selection_codes() RETURNS integer LANGUAGE sql AS $function$
  UPDATE bet.prediction_bet pb SET selection_code='exact_score',updated_at=now() FROM bet.market_type mt WHERE mt.market_type_id=pb.market_type_id AND mt.market_type_code='correct_score' AND pb.selection_code='exact';
  SELECT COUNT(*)::integer FROM bet.prediction_bet pb JOIN bet.market_type mt ON mt.market_type_id=pb.market_type_id WHERE mt.market_type_code='correct_score' AND pb.selection_code='exact_score';
$function$;
