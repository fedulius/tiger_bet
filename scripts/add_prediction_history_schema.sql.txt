-- Tiger Bet: expandable prediction history / settlement schema
-- Date: 2026-07-15
-- Purpose:
--   1) keep public.match_analysis as immutable analysis snapshot;
--   2) add analysis_type to classify analysis pipelines;
--   3) add bet.* publication/settlement layer for daily, VIP, live, express and future strategy flows;
--   4) normalize every recommended bet into independently settleable rows.
--
-- IMPORTANT:
--   This file is a proposal/migration draft. Review names and run in staging first.
--   It does not modify existing data except adding/backfilling match_analysis.analysis_type_id.
--
-- Core model:
--   public.match_analysis       = what the analytics pipeline calculated
--   bet.prediction_card         = what was published to users
--   bet.prediction_card_source  = which analyses were used by the card
--   bet.prediction_bet          = one single bet, or one express bet container
--   bet.prediction_bet_leg      = legs inside express/combined bets
--   bet.bet_settlement          = current settlement for a bet
--   bet.bet_leg_settlement      = current settlement for an express leg
--   bet.bet_settlement_revision = audit log for recalculations/manual fixes

BEGIN;

-- ============================================================
-- 1. Analysis type for public.match_analysis
-- ============================================================

CREATE TABLE IF NOT EXISTS public.analysis_type (
  analysis_type_id integer GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  analysis_type_code varchar(64) NOT NULL UNIQUE,
  analysis_type_name varchar(160) NOT NULL,
  is_active boolean NOT NULL DEFAULT true,
  created_at timestamptz NOT NULL DEFAULT now(),
  updated_at timestamptz NULL
);

INSERT INTO public.analysis_type (analysis_type_code, analysis_type_name)
VALUES
  ('daily_pick', 'Daily pick analysis'),
  ('vip_single', 'VIP single-match analysis'),
  ('live_single', 'Live single-match analysis'),
  ('manual_single', 'Manual single-match analysis'),
  ('strategy_candidate', 'Strategy candidate analysis'),
  ('backtest', 'Backtest analysis')
ON CONFLICT (analysis_type_code) DO NOTHING;

ALTER TABLE public.match_analysis
  ADD COLUMN IF NOT EXISTS analysis_type_id integer;

UPDATE public.match_analysis
SET analysis_type_id = (
  SELECT analysis_type_id
  FROM public.analysis_type
  WHERE analysis_type_code = 'daily_pick'
)
WHERE analysis_type_id IS NULL;

ALTER TABLE public.match_analysis
  ALTER COLUMN analysis_type_id SET NOT NULL;

ALTER TABLE public.match_analysis
  ADD CONSTRAINT match_analysis_analysis_type_analysis_type_id_fk
  FOREIGN KEY (analysis_type_id)
  REFERENCES public.analysis_type(analysis_type_id)
  NOT VALID;

ALTER TABLE public.match_analysis
  VALIDATE CONSTRAINT match_analysis_analysis_type_analysis_type_id_fk;

CREATE INDEX IF NOT EXISTS match_analysis_analysis_type_id_idx
  ON public.match_analysis (analysis_type_id);

-- ============================================================
-- 4. Prediction bets and express legs
-- ============================================================

CREATE TABLE IF NOT EXISTS bet.prediction_bet (
  prediction_bet_id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  prediction_card_id bigint NOT NULL REFERENCES bet.prediction_card(prediction_card_id) ON DELETE RESTRICT,

  -- single = normal independently settleable bet; express = container with legs.
  kind varchar(32) NOT NULL DEFAULT 'single',
  ordinal smallint NOT NULL,

  -- For single bets. Express containers normally keep match/market fields NULL and use prediction_bet_leg.
  match_id integer NULL REFERENCES public.match(match_id),
  match_analysis_id integer NULL REFERENCES public.match_analysis(match_analysis_id),

  market_type_id smallint NULL REFERENCES bet.market_type(market_type_id),
  period_id smallint NULL REFERENCES bet.period(period_id),
  participant_scope varchar(64) NULL,
  selection_code varchar(80) NULL,
  line_value numeric(10,3) NULL,
  score_home smallint NULL,
  score_away smallint NULL,

  market_key text NOT NULL,
  display_label text NOT NULL,
  odds_decimal numeric(10,4) NULL,

  risk_level_id smallint NULL REFERENCES bet.risk_level(risk_level_id),
  confidence numeric(5,2) NULL,
  reason text NULL,

  source_payload jsonb NULL,
  selection_snapshot jsonb NOT NULL DEFAULT '{}'::jsonb,

  created_at timestamptz NOT NULL DEFAULT now(),
  updated_at timestamptz NULL,

  CONSTRAINT prediction_bet_kind_chk CHECK (kind IN ('single', 'express')),
  CONSTRAINT prediction_bet_ordinal_chk CHECK (ordinal > 0),
  CONSTRAINT prediction_bet_odds_decimal_chk CHECK (odds_decimal IS NULL OR odds_decimal >= 1.0000),
  CONSTRAINT prediction_bet_confidence_chk CHECK (confidence IS NULL OR confidence BETWEEN 0 AND 100),
  CONSTRAINT prediction_bet_participant_scope_chk CHECK (
    participant_scope IS NULL OR participant_scope IN ('match', 'home_team', 'away_team', 'player')
  ),
  CONSTRAINT prediction_bet_source_payload_object_chk CHECK (source_payload IS NULL OR jsonb_typeof(source_payload) = 'object'),
  CONSTRAINT prediction_bet_selection_snapshot_object_chk CHECK (jsonb_typeof(selection_snapshot) = 'object'),

  -- Single bets need enough normalized fields to settle/filter.
  -- Express containers should be settled through legs and can keep these fields NULL.
  CONSTRAINT prediction_bet_single_fields_chk CHECK (
    kind = 'express'
    OR (
      kind = 'single'
      AND match_id IS NOT NULL
      AND market_type_id IS NOT NULL
      AND period_id IS NOT NULL
      AND participant_scope IS NOT NULL
      AND selection_code IS NOT NULL
    )
  ),

  CONSTRAINT prediction_bet_card_ordinal_uniq UNIQUE (prediction_card_id, ordinal)
);

COMMENT ON TABLE bet.prediction_bet IS
  'One recommendation inside a published card. Daily MVP creates three single rows from match_analysis.recommended_bets. Express cards create one express row plus legs.';

COMMENT ON COLUMN bet.prediction_bet.market_key IS
  'Canonical machine key, e.g. total|full_time|match|over|2.5 or handicap|full_time|home_team|home|-1.5.';

CREATE INDEX IF NOT EXISTS prediction_bet_card_idx
  ON bet.prediction_bet (prediction_card_id, ordinal);

CREATE INDEX IF NOT EXISTS prediction_bet_match_idx
  ON bet.prediction_bet (match_id);

CREATE INDEX IF NOT EXISTS prediction_bet_analysis_idx
  ON bet.prediction_bet (match_analysis_id);

CREATE INDEX IF NOT EXISTS prediction_bet_market_idx
  ON bet.prediction_bet (market_type_id);

CREATE INDEX IF NOT EXISTS prediction_bet_period_idx

CREATE INDEX IF NOT EXISTS prediction_bet_risk_idx
  ON bet.prediction_bet (risk_level_id);

CREATE INDEX IF NOT EXISTS prediction_bet_market_key_idx

CREATE TABLE IF NOT EXISTS bet.bet_settlement (
  bet_settlement_id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  prediction_bet_id bigint NOT NULL UNIQUE REFERENCES bet.prediction_bet(prediction_bet_id) ON DELETE RESTRICT,

  settlement_status_id smallint NOT NULL REFERENCES bet.settlement_status(settlement_status_id),
  settlement_result_id smallint NULL REFERENCES bet.settlement_result(settlement_result_id),

  settled_at timestamptz NULL,
  settlement_version integer NOT NULL DEFAULT 1,
  rule_version varchar(80) NOT NULL DEFAULT 'v1',

  score_scope jsonb NOT NULL DEFAULT '{}'::jsonb,
  outcome_value jsonb NULL,
  profit_factor numeric(12,6) NULL,

  reason_code varchar(120) NULL,
  reason_text text NULL,

  calculated_at timestamptz NOT NULL DEFAULT now(),
  calculated_by varchar(80) NOT NULL DEFAULT 'rule_engine',
  raw_payload jsonb NULL,

  CONSTRAINT bet_settlement_version_chk CHECK (settlement_version > 0),
  CONSTRAINT bet_settlement_score_scope_object_chk CHECK (jsonb_typeof(score_scope) = 'object'),
  CONSTRAINT bet_settlement_outcome_value_json_chk CHECK (outcome_value IS NULL OR jsonb_typeof(outcome_value) IN ('object', 'number', 'string', 'boolean')),
  CONSTRAINT bet_settlement_raw_payload_object_chk CHECK (raw_payload IS NULL OR jsonb_typeof(raw_payload) = 'object'),
  CONSTRAINT bet_settlement_calculated_by_chk CHECK (calculated_by IN ('rule_engine', 'manual', 'provider', 'backfill'))
);

COMMENT ON TABLE bet.bet_settlement IS

  prediction_bet_id bigint NULL REFERENCES bet.prediction_bet(prediction_bet_id) ON DELETE RESTRICT,
  prediction_bet_leg_id bigint NULL REFERENCES bet.prediction_bet_leg(prediction_bet_leg_id) ON DELETE RESTRICT,

  previous_settlement_status_id smallint NULL REFERENCES bet.settlement_status(settlement_status_id),
  previous_settlement_result_id smallint NULL REFERENCES bet.settlement_result(settlement_result_id),

    (prediction_bet_id IS NOT NULL AND prediction_bet_leg_id IS NULL)

  CONSTRAINT bet_settlement_revision_payload_object_chk CHECK (jsonb_typeof(payload) = 'object'),
  CONSTRAINT bet_settlement_revision_changed_by_chk CHECK (changed_by IN ('rule_engine', 'manual', 'provider', 'backfill'))

  'Append-only audit trail for settlement recalculations/manual fixes/provider corrections.';

  -- saved = bookmarked; placed = user says they placed it; skipped/dismissed are product signals; removed = soft delete.
  decision_code varchar(40) NOT NULL,
  decision_at timestamptz NOT NULL DEFAULT now(),

  stake_amount numeric(14,2) NULL,
  stake_currency char(3) NULL,
  user_odds_decimal numeric(10,4) NULL,
  placed_at timestamptz NULL,

  note text NULL,
  meta jsonb NOT NULL DEFAULT '{}'::jsonb,

  created_at timestamptz NOT NULL DEFAULT now(),
  updated_at timestamptz NULL,

  CONSTRAINT user_pick_user_odds_chk CHECK (user_odds_decimal IS NULL OR user_odds_decimal >= 1.0000),
  CONSTRAINT user_pick_meta_object_chk CHECK (jsonb_typeof(meta) = 'object')
);

CREATE OR REPLACE VIEW bet.v_prediction_bet_settlements AS
SELECT
  pb.*,
  bs.bet_settlement_id,
  bs.settlement_status_id,
  ss.settlement_status_code,
  ss.settlement_status_name,
  ss.is_terminal AS settlement_is_terminal,
  bs.settlement_result_id,
  sr.settlement_result_code,
  sr.settlement_result_name,
  bs.settled_at,
  bs.settlement_version,
  bs.rule_version,
  bs.score_scope,
  bs.outcome_value,
  bs.profit_factor,
  bs.reason_code,
  bs.reason_text,
  bs.calculated_at,
  bs.calculated_by,
  bs.raw_payload,
  CASE
    WHEN ss.settlement_status_code = 'pending' THEN 'pending'
    WHEN ss.settlement_status_code = 'not_supported' THEN 'not_supported'
    WHEN ss.settlement_status_code = 'cancelled' THEN 'cancelled'
    WHEN ss.settlement_status_code = 'void' THEN 'void'
    WHEN sr.settlement_result_code = 'win' THEN 'won'
    WHEN sr.settlement_result_code = 'loss' THEN 'lost'
    WHEN sr.settlement_result_code IN ('push', 'void') THEN 'void'
    WHEN sr.settlement_result_code IN ('half_win', 'half_loss') THEN sr.settlement_result_code
    ELSE 'unknown'
  END AS ui_result_code,
  CASE
    WHEN ss.settlement_status_code = 'pending' THEN 'Ждём результат'
    WHEN ss.settlement_status_code = 'not_supported' THEN 'Пока не рассчитываем'
    WHEN ss.settlement_status_code = 'cancelled' THEN 'Снято'
    WHEN ss.settlement_status_code = 'void' THEN 'Возврат'
    WHEN sr.settlement_result_code = 'win' THEN 'Зашло'
    WHEN sr.settlement_result_code = 'loss' THEN 'Не зашло'
    WHEN sr.settlement_result_code IN ('push', 'void') THEN 'Возврат'
    WHEN sr.settlement_result_code = 'half_win' THEN 'Половина выигрыша'
    WHEN sr.settlement_result_code = 'half_loss' THEN 'Половина проигрыша'
    ELSE 'Не определено'
  END AS ui_result_label
FROM bet.v_prediction_bets pb
LEFT JOIN bet.bet_settlement bs ON bs.prediction_bet_id = pb.prediction_bet_id
LEFT JOIN bet.settlement_status ss ON ss.settlement_status_id = bs.settlement_status_id
LEFT JOIN bet.settlement_result sr ON sr.settlement_result_id = bs.settlement_result_id;

COMMENT ON VIEW bet.v_prediction_bet_settlements IS
  'Main /history read model: every bet with current settlement and UI-friendly result labels.';

CREATE OR REPLACE VIEW bet.v_prediction_card_history AS
SELECT
  pc.prediction_card_id,
  pc.card_type_code,
  pc.card_status_code,
  pc.visibility_scope,
  pc.published_at,
  pc.published_date,
  pc.primary_match_id,
  pc.home_team,
  pc.away_team,
  pc.match_start_at,
  pc.tournament_id,
  pc.tournament_name,
  pc.title,
  pc.headline,
  COUNT(pb.prediction_bet_id) AS bets_count,
  COUNT(pb.prediction_bet_id) FILTER (WHERE pb.kind = 'single') AS single_bets_count,
  COUNT(pb.prediction_bet_id) FILTER (WHERE pb.kind = 'express') AS express_bets_count,
  COUNT(pb.prediction_bet_id) FILTER (WHERE pbs.settlement_status_code = 'pending' OR pbs.settlement_status_code IS NULL) AS pending_count,
  COUNT(pb.prediction_bet_id) FILTER (WHERE pbs.settlement_result_code = 'win') AS won_count,
  COUNT(pb.prediction_bet_id) FILTER (WHERE pbs.settlement_result_code = 'loss') AS lost_count,
  COUNT(pb.prediction_bet_id) FILTER (WHERE pbs.settlement_result_code IN ('push', 'void')) AS void_count,
  COUNT(pb.prediction_bet_id) FILTER (WHERE pbs.settlement_status_code = 'not_supported') AS not_supported_count,
  SUM(COALESCE(pbs.profit_factor, 0)) AS profit_units,
  CASE
    WHEN COUNT(pb.prediction_bet_id) FILTER (WHERE pbs.settlement_result_code IN ('win', 'loss')) = 0 THEN NULL
    ELSE ROUND(
      100.0 * COUNT(pb.prediction_bet_id) FILTER (WHERE pbs.settlement_result_code = 'win')
      / NULLIF(COUNT(pb.prediction_bet_id) FILTER (WHERE pbs.settlement_result_code IN ('win', 'loss')), 0),
      2
    )
  END AS hit_rate_percent
FROM bet.v_prediction_cards pc
LEFT JOIN bet.prediction_bet pb ON pb.prediction_card_id = pc.prediction_card_id
LEFT JOIN bet.v_prediction_bet_settlements pbs ON pbs.prediction_bet_id = pb.prediction_bet_id
GROUP BY
  pc.prediction_card_id,
  pc.card_type_code,
  pc.card_status_code,
  pc.visibility_scope,
  pc.published_at,
  pc.published_date,
  pc.primary_match_id,
  pc.home_team,
  pc.away_team,
  pc.match_start_at,
  pc.tournament_id,
  pc.tournament_name,
  pc.title,
  pc.headline;

COMMENT ON VIEW bet.v_prediction_card_history IS
  'One row per card with bet counts, current outcomes, profit units and hit rate. Good for list-level /history cards.';

CREATE OR REPLACE VIEW bet.v_performance_by_market AS
SELECT
  market_type_id,
  market_type_code,
  market_type_name,
  period_id,
  period_code,
  participant_scope,
  risk_level_id,
  risk_level_code,
  COUNT(*) AS bets_count,
  COUNT(*) FILTER (WHERE settlement_status_code = 'settled') AS settled_count,
  COUNT(*) FILTER (WHERE settlement_status_code = 'pending' OR settlement_status_code IS NULL) AS pending_count,
  COUNT(*) FILTER (WHERE settlement_status_code = 'not_supported') AS not_supported_count,
  COUNT(*) FILTER (WHERE settlement_result_code = 'win') AS won_count,
  COUNT(*) FILTER (WHERE settlement_result_code = 'loss') AS lost_count,
  COUNT(*) FILTER (WHERE settlement_result_code IN ('push', 'void')) AS void_count,
  AVG(odds_decimal) FILTER (WHERE odds_decimal IS NOT NULL) AS avg_odds,
  SUM(COALESCE(profit_factor, 0)) AS profit_units,
  CASE
    WHEN COUNT(*) FILTER (WHERE settlement_result_code IN ('win', 'loss')) = 0 THEN NULL
    ELSE ROUND(
      100.0 * COUNT(*) FILTER (WHERE settlement_result_code = 'win')
      / NULLIF(COUNT(*) FILTER (WHERE settlement_result_code IN ('win', 'loss')), 0),
      2
    )
  END AS hit_rate_percent
FROM bet.v_prediction_bet_settlements
WHERE kind = 'single'
  AND card_status_code = 'published'
GROUP BY
  market_type_id,
  market_type_code,
  market_type_name,
  period_id,
  period_code,
  participant_scope,
  risk_level_id,
  risk_level_code;

COMMENT ON VIEW bet.v_performance_by_market IS
  'Tiger Bet product performance grouped by market/period/scope/risk. not_supported is tracked separately and not counted as loss.';

-- 8. Compatibility / migration notes
-- ============================================================
--
-- Recommended backfill flow:
--   1) Select published/ready public.match_analysis rows that actually appeared in daily picks.
--   2) For each row create bet.prediction_card:
--        card_type=daily, status=published, primary_match_id from match_source.match_id,
--        primary_match_analysis_id=match_analysis_id,
--        title/headline/brief/risk_note from match_analysis,
--        publication_hash deterministic from analysis_hash + card_type + edition.
--   3) Create bet.prediction_card_source for the analysis.
--   4) Expand recommended_bets jsonb array into bet.prediction_bet rows:
--        ordinal = array index + 1,
--        source JSON -> selection_snapshot,
--        type/outcome -> market_type/selection/scope/line/market_key.
--   5) Create bet.bet_settlement rows with status=pending.
--   6) Run settlement worker. Unsupported markets become not_supported/unknown, not loss.
--
-- Current observed recommended_bets normalization examples:
--   one_x_two / w1          -> one_x_two | full_time | match     | home       | NULL
--   one_x_two / w2          -> one_x_two | full_time | match     | away       | NULL
--   double_chance / x2      -> double_chance | full_time | match | away_or_draw | NULL
--   both_to_score / yes     -> both_to_score | full_time | match | yes        | NULL
--   total_over / 2_5        -> total | full_time | match         | over       | 2.5
--   total_t2_over / 1_5     -> team_total | full_time | away_team | over       | 1.5
--   handicap1 / -1_5        -> handicap | full_time | home_team   | home       | -1.5
--   correct_score / 1:2     -> correct_score | full_time | match | exact_score | score_home=1, score_away=2
--   corners_total_over / 9_5 -> corners_total | full_time | match | over      | 9.5
--
-- History UI should read:
--   bet.prediction_card
--   JOIN bet.prediction_bet
--   LEFT JOIN bet.bet_settlement
--   JOIN public.match / public.tournament through prediction_bet.match_id or card.primary_match_id.
--
-- Tiger Bet product statistics should count bet.prediction_bet rows attached to published cards.
-- User statistics should count bet.user_pick / future real user_bet separately.
